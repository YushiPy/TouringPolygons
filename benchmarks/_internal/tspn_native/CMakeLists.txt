cmake_minimum_required(VERSION 3.23)
project(tpp_tspn_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_BUILD_TYPE Release)
get_filename_component(TPP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FEKETE_SOURCE "${TPP_ROOT}/third_party/tspn-socg" CACHE PATH "Pinned Fekete checkout (read only)")
set(GUROBI_HOME "/Library/gurobi1303/macos_universal2" CACHE PATH "Gurobi installation")
if(NOT EXISTS "${FEKETE_SOURCE}/tspn_core/CMakeLists.txt")
    message(FATAL_ERROR "Set FEKETE_SOURCE to an initialized tspn-socg checkout")
endif()
add_subdirectory("${TPP_ROOT}/packages/nonconvex-tpp/cpp" touring_polygons)
find_package(CGAL REQUIRED CONFIG)
find_package(Boost REQUIRED CONFIG COMPONENTS thread)
find_package(fmt REQUIRED CONFIG)
find_package(nlohmann_json QUIET CONFIG)
if(NOT TARGET nlohmann_json::nlohmann_json)
    find_path(NLOHMANN_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
    add_library(nlohmann_json::nlohmann_json INTERFACE IMPORTED)
    target_include_directories(nlohmann_json::nlohmann_json INTERFACE "${NLOHMANN_INCLUDE_DIR}")
endif()
file(GLOB GUROBI_LIBRARY "${GUROBI_HOME}/lib/libgurobi[0-9]*.dylib" "${GUROBI_HOME}/lib/libgurobi[0-9]*.so")
list(FILTER GUROBI_LIBRARY EXCLUDE REGEX "_light")
list(GET GUROBI_LIBRARY 0 GUROBI_LIBRARY)
add_library(gurobi::gurobi INTERFACE IMPORTED)
target_include_directories(gurobi::gurobi INTERFACE "${GUROBI_HOME}/include")
target_link_libraries(gurobi::gurobi INTERFACE "${GUROBI_LIBRARY}" "${GUROBI_HOME}/lib/libgurobi_c++.a")
set(CGAL_TARGET CGAL::CGAL)
# The fork's optional path oracle links our existing library; the benchmark
# selects its original SOCP backend. Build its own CMake source list unchanged.
add_library(tpp_embedded_oracle ALIAS tpp_convex)
add_subdirectory("${FEKETE_SOURCE}/tspn_core" fekete)
target_include_directories(tspn_core PUBLIC "${FEKETE_SOURCE}")
target_link_libraries(tspn_core PRIVATE fmt::fmt nlohmann_json::nlohmann_json Boost::thread)
target_compile_options(tspn_core PRIVATE -O3)
add_executable(tpp-fekete-cycle main.cpp)
target_link_libraries(tpp-fekete-cycle PRIVATE tspn_core fmt::fmt nlohmann_json::nlohmann_json CGAL::CGAL)
enable_testing()
add_executable(tpp-fekete-termination-test termination_test.cpp)
add_test(NAME tpp-fekete-termination-test COMMAND tpp-fekete-termination-test)
