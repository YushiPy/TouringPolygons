cmake_minimum_required(VERSION 3.20)
project(tpp-convex)

set(TPP_CXX_STANDARD 26 CACHE STRING "C++ standard used by the maintained TPP packages")
set_property(CACHE TPP_CXX_STANDARD PROPERTY STRINGS 23 26)
set(CMAKE_CXX_STANDARD ${TPP_CXX_STANDARD})

set(TARGET "main" CACHE STRING "Target to build")
option(TPP_ENABLE_GUROBI "Build the optional Gurobi baseline solver" OFF)
option(TPP_ENABLE_GMP_RATIONAL "Use GMP for exact arithmetic when available (native builds)" ON)

# Both backends implement the same unbounded rational operations. Keep the
# header-only backend available for portable builds, including WebAssembly.
add_library(tpp_exact_arithmetic INTERFACE)
option(TPP_FAST_INTERVAL_ROUNDING "Expand binary64 intervals using adjacent IEEE bit patterns" ON)
if(TPP_FAST_INTERVAL_ROUNDING)
    target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_HAS_FAST_INTERVAL_ROUNDING=1)
endif()
option(TPP_DYADIC_MEMBERSHIP "Resolve ambiguous binary orientations with bounded exact integers" ON)
if(TPP_DYADIC_MEMBERSHIP)
    target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_HAS_DYADIC_MEMBERSHIP=1)
endif()
option(TPP_MEMO_BINARY_MEMBERSHIP "Reuse two binary contact-membership proofs per prepared polygon" ON)
if(TPP_MEMO_BINARY_MEMBERSHIP)
    target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_HAS_MEMO_BINARY_MEMBERSHIP=1)
endif()
option(TPP_DENSE_PAIR_CACHE "Store prepared pair proofs for the first 256 identities in bit tables" ON)
if(TPP_DENSE_PAIR_CACHE)
    target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_HAS_DENSE_PAIR_CACHE=1)
endif()
option(TPP_DENSE_DISJOINT_SUBSETS "Dispatch sequences using bit masks of already proved disjoint pairs" ON)
if(TPP_DENSE_PAIR_CACHE AND TPP_DENSE_DISJOINT_SUBSETS)
    target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_HAS_DENSE_DISJOINT_SUBSETS=1)
endif()
option(TPP_KKT_STRAIGHT_FIRST "Skip rational squared norms for zero and straight KKT links" OFF)
if(TPP_KKT_STRAIGHT_FIRST)
    target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_HAS_KKT_STRAIGHT_FIRST=1)
endif()
if(TPP_ENABLE_GMP_RATIONAL AND NOT EMSCRIPTEN)
    find_path(TPP_GMP_INCLUDE_DIR gmp.h HINTS /opt/homebrew/include /opt/homebrew/opt/gmp/include)
    find_library(TPP_GMP_LIBRARY NAMES gmp HINTS /opt/homebrew/lib /opt/homebrew/opt/gmp/lib)
    if(TPP_GMP_INCLUDE_DIR AND TPP_GMP_LIBRARY)
        target_compile_definitions(tpp_exact_arithmetic INTERFACE TPP_USE_GMP_RATIONAL=1)
        target_include_directories(tpp_exact_arithmetic INTERFACE ${TPP_GMP_INCLUDE_DIR})
        target_link_libraries(tpp_exact_arithmetic INTERFACE ${TPP_GMP_LIBRARY})
        message(STATUS "TPP exact arithmetic: GMP")
    else()
        message(STATUS "GMP unavailable; TPP exact arithmetic: Boost cpp_rational")
    endif()
else()
    message(STATUS "TPP exact arithmetic: Boost cpp_rational")
endif()

if(NOT TARGET tpp_geometry)
    add_subdirectory(../../common-geometry/cpp tpp_geometry)
endif()

if(APPLE)
    set(OpenMP_ROOT /opt/homebrew/opt/libomp)
    set(OpenMP_INCLUDE_DIR /opt/homebrew/opt/libomp/include)

    set(OpenMP_C_FLAGS "-Xpreprocessor -fopenmp")
    set(OpenMP_CXX_FLAGS "-Xpreprocessor -fopenmp")

    set(OpenMP_C_LIB_NAMES "omp")
    set(OpenMP_CXX_LIB_NAMES "omp")

    set(OpenMP_omp_LIBRARY /opt/homebrew/opt/libomp/lib/libomp.dylib)
endif()

file(GLOB_RECURSE TPP_CONVEX_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER TPP_CONVEX_SOURCES EXCLUDE REGEX "src/main.*\\.cpp")

if(NOT TPP_ENABLE_GUROBI)
    list(FILTER TPP_CONVEX_SOURCES EXCLUDE REGEX "src/solvers/gurobi\\.cpp")
endif()

find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Boost REQUIRED CONFIG)
add_library(tpp_directional_double OBJECT src/solvers/intersecting_maps.cpp)
option(TPP_FILTERED_DIRECTIONAL "Recover rejected directional proposals with lazy exact rationals" ON)
if(TPP_FILTERED_DIRECTIONAL)
    add_library(tpp_directional_filtered OBJECT src/solvers/intersecting_maps.cpp)
    target_compile_definitions(tpp_directional_filtered PRIVATE
        TPP_EXPERIMENT_NATIVE_DOUBLE=1 TPP_DIRECTIONAL_FILTERED=1)
    target_include_directories(tpp_directional_filtered PRIVATE include)
    target_link_libraries(tpp_directional_filtered PRIVATE Boost::headers tpp_geometry tpp_exact_arithmetic)
    target_compile_options(tpp_directional_filtered PRIVATE -O3)
endif()
target_compile_definitions(tpp_directional_double PRIVATE
    TPP_EXPERIMENT_NATIVE_DOUBLE=1
    TPP_DIRECTIONAL_DOUBLE_VARIANT=1)
target_include_directories(tpp_directional_double PRIVATE include)
target_link_libraries(tpp_directional_double PRIVATE Boost::headers tpp_geometry tpp_exact_arithmetic)
target_compile_options(tpp_directional_double PRIVATE -O3)
add_library(tpp_convex STATIC ${TPP_CONVEX_SOURCES})
option(TPP_INTERVAL_PRIMAL_DUAL "Try certified interval bounds before rational replay" ON)
if(TPP_INTERVAL_PRIMAL_DUAL)
    target_compile_definitions(tpp_convex PUBLIC TPP_HAS_INTERVAL_PRIMAL_DUAL=1)
endif()
option(TPP_TOUCHING_DISJOINT "Try certified disjoint traces for boundary contacts" ON)
if(TPP_TOUCHING_DISJOINT)
    target_compile_definitions(tpp_convex PUBLIC TPP_HAS_TOUCHING_DISJOINT=1)
endif()
target_sources(tpp_convex PRIVATE $<TARGET_OBJECTS:tpp_directional_double>)
if(TPP_FILTERED_DIRECTIONAL)
    target_sources(tpp_convex PRIVATE $<TARGET_OBJECTS:tpp_directional_filtered>)
    target_compile_definitions(tpp_convex PUBLIC TPP_HAS_FILTERED_DIRECTIONAL=1)
endif()
target_link_libraries(tpp_convex PRIVATE Eigen3::Eigen PUBLIC Boost::headers tpp_exact_arithmetic)
target_include_directories(tpp_convex PUBLIC include)
target_link_libraries(tpp_convex PUBLIC tpp_geometry)
target_compile_options(tpp_convex PRIVATE -O3)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/${TARGET}.cpp")
    add_executable(${PROJECT_NAME} src/${TARGET}.cpp)
    target_link_libraries(${PROJECT_NAME} PRIVATE tpp_convex Boost::headers)
    target_compile_options(${PROJECT_NAME} PRIVATE -O3)
    if("${TARGET}" STREQUAL "main-cycle_tests")
        target_compile_definitions(${PROJECT_NAME} PRIVATE
            TPP_CYCLE_REGRESSION_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/cycle_active_contacts.json"
            TPP_CYCLE_REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../../benchmarks/results-saved/convex-cycle-gurobi-reference-2026-09-25")
    endif()
endif()

find_package(OpenMP REQUIRED)
target_link_libraries(tpp_convex PUBLIC OpenMP::OpenMP_CXX)

if(APPLE)
    target_include_directories(tpp_convex PUBLIC ${OpenMP_INCLUDE_DIR})
endif()

if(TPP_ENABLE_GUROBI)
    target_compile_definitions(tpp_convex PUBLIC TPP_ENABLE_GUROBI=1)

    if(DEFINED GUROBI_HOME)
        set(GUROBI_HOME ${GUROBI_HOME})
    elseif(DEFINED ENV{GUROBI_HOME})
        set(GUROBI_HOME $ENV{GUROBI_HOME})
    else()
        message(FATAL_ERROR "GUROBI_HOME not set")
    endif()

    target_include_directories(tpp_convex PUBLIC ${GUROBI_HOME}/include)
    target_link_directories(tpp_convex PRIVATE ${GUROBI_HOME}/lib)

    file(GLOB GUROBI_LIB "${GUROBI_HOME}/lib/libgurobi[0-9]*.dylib")
    list(FILTER GUROBI_LIB EXCLUDE REGEX ".*_light.*")
    if(NOT GUROBI_LIB)
        message(FATAL_ERROR "Could not find versioned Gurobi shared library in ${GUROBI_HOME}/lib")
    endif()
    list(GET GUROBI_LIB 0 GUROBI_LIB)

    target_link_libraries(tpp_convex PRIVATE
        ${GUROBI_LIB}
        ${GUROBI_HOME}/lib/libgurobi_c++.a
    )
endif()
